//---------------------/---------------------
//           Find Best Hero
//---------------------/---------------------
function findBestAttackHero(){
  var totA:Number=0, heroAtt:number=0, heroName:String, cityName:String, heroId:Number;
  for each(i in c.ac){
    if (i.cm.bestAttackHero()==null) totA = 0;
    else totA = (i.cm.bestAttackHero().attack + i.cm.bestAttackHero().power)* (1 + i.cm.bestAttackHero().powerBuffAdded/100);
    if (totA>heroAtt){
      heroName = i.cm.bestAttackHero().name;
      heroId = i.cm.bestAttackHero().id;
      cityName = i.castle.name;
      heroAtt = totA;}}
  setVar("hero",  heroName);
  setVar("heroId",  heroId);
  msg = "The best Attack hero in this account, <b>" +heroName+ ": " + heroAtt + " atk</b>, is currently stationed in " +cityName;
  return msg;}
//---------------------/---------------------
//    Find City in Turn
//---------------------/---------------------
function cityRipeForQing():Boolean{
  var ripe:Boolean, trainer:String;
  for each(j in c.ac){
    if (j.isHeroInCastle(getVar('hero'))){
      trainer = getVar('hero') + " is currently present in <b>" +j.castle.name+ "</b> where the population is at <b>" + c.formatNum(j.cm.resource.curPopulation/j.cm.resource.maxPopulation*100,1) + "&#37;</b>.";
      if (j.cm.resource.curPopulation==j.cm.resource.maxPopulation){
        setVar("c", c.ac.indexOf(j));
        setVar("id", j.castle.id);
        setVar("prod", j.cm.getVar('stdProd'));
        ripe = true;}
      else {
        ripe = false;}}}
  c.cm.logMsg(trainer);
  return ripe;}
//---------------------/---------------------
//   Shortest Troop Queue
//---------------------/---------------------
function getShortestTroopQueue(x){
  var barracks:Array = new Array(), batchesPerBarrack:Array = new Array();
  c.ac[x].cm.updateTroopQueue();
  for each (ivn in c.ac[x].cm.troopProduceQueue){
    barracks.push(ivn.positionId);
    batchesPerBarrack.push(ivn.allProduceQueueArray.length)}
  c.af.getInteriorCommands().modifyTaxRate(c.ac[x].castle.id,0);
  msg = "-----------------------------------------------------------------------------------------------------------------------------<br><tab><tab><tab><tab>   <tab><tab><tab><tab>Population in " +c.ac[x].castle.name+": "+c.ac[x].cm.resource.curPopulation+ " of " +c.ac[x].cm.resource.maxPopulation+ "<br><tab><tab><tab><tab>  <tab><tab><tab>   Current number of batches in the barracks";
  c.cm.logMsg(msg);
  msg = "<font color='#0B610B'><b>Barracks Q in <font color='#990050'><u><i>" +c.ac[x].castle.name+ "</i></u></font>: " +batchesPerBarrack.join(" ")+ "</b></font><br><tab><tab><tab><tab> -----------------------------------------------------------------------------------------------------------------------------";
  c.cm.warLogMsg(msg);
  setVar("maxQ", Math.max.apply(null,batchesPerBarrack));
  setVar("minQ", Math.min.apply(null,batchesPerBarrack));
  return barracks[batchesPerBarrack.indexOf(Math.min.apply(null,batchesPerBarrack))];}
//---------------------/---------------------
//        Food Availability
//---------------------/---------------------
function locateZeroFood(x):String{
  var valleyFood:Number = 0, valleyFoodArmyId:int, valleyFoodArmyDirection:int, valleyFoodArmyEta:int, wait:String=false, msg:String; 
  setVar("eta", 0);
  if (c.ac[x].cm.localArmies.length==0) valleyFood = c.cm.resource.food.amount;
  else {
    for each (vFI in c.ac[x].cm.localArmies){
      if (vFI.resource.food>valleyFood&&vFI.targetFieldId==c.ac[x].cm.fields[0].id){
        valleyFoodArmyId = vFI.armyId;
        valleyFood = vFI.resource.food;
        valleyFoodArmyDirection = vFI.direction;
        valleyFoodArmyEta = vFI.reachTime;}}
    if (valleyFood==0) valleyFood = c.cm.resource.food.amount;
    else if (valleyFoodArmyDirection==2){
      wait = true;
      c.cm.logMsg("Waiting for food to return, exected arrival <b>"+c.getDateFromTime(valleyFoodArmyEta).toFormattedTimeString()+"</b>");
      setVar("eta", valleyFoodArmyEta);}}
  if (valleyFood>=1000000000) msg = "<font color='#2E64FE'><b>" +c.ac[x].castle.name+ "</b> has <b>" +c.formatNum(valleyFood/1000000000,0)+ " bil</b> Food";
  else if (valleyFood>100000000) msg = "<b>" +c.ac[x].castle.name +"</b> has <b>" +c.formatNum(valleyFood/1000000,0)+ " mil</b> Food";
  else msg = "<font color='#FF0000'><b>" +c.ac[x].castle.name+ "</b> has only <b>" +c.formatNum(valleyFood/1000000,0)+ " mil</b> Food"; 
  c.cm.logMsg(msg);
  setVar("vF", valleyFood);
  setVar("foodArmyId", valleyFoodArmyId);
  return wait;}
//---------------------/---------------------
//  Find Next Active City
//---------------------/---------------------
function findNextCity(next){
  if (++next == c.ac.length) next = 0;
  if (c.ac[next].cm.getVar('trainingActive')=="false" && next==c.ac.length) next = 0;
  return c.ac[next].cm.getVar('trainingActive')=="false" ? findNextCity(next) : next;}
//---------------------/---------------------
//         List Active Cities
//---------------------/---------------------
function whereIsTroopTrainingActive():String{
var activeCities:String = "Trooptraining is active in: <b>", no:int; 
  for each (h in  c.ac){
    if (h.cm.getVar("trainingActive")=="true"){activeCities += h.castle.name + "  ";no += 1;}}
  activeCities += "</b>Each city is visited with " + no/2 + " hour(s) interval.";
 return activeCities;}
