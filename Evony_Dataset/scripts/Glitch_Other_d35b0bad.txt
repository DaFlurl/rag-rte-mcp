import flash.events.Event;
import flash.events.ProgressEvent;
import autoevony.common.Utils;
import flash.net.Socket;
import flash.utils.ByteArray;
function datahandler(event:ProgressEvent):int{
c.cm.logMsg("Received message.");
var sock:Socket = c.m_techLevels.sock;
var fo:ByteArray = new ByteArray();
var by:int = sock.bytesAvailable;
sock.readBytes(fo,0,by);
var amount:int = fo.readObject();
if (c.m_techLevels.lastupdate!=c.m_techLevels.lastsold){
c.m_techLevels.totalsold=0;
c.m_techLevels.lastsold=c.m_techLevels.lastupdate;
}
var total:int=c.m_techLevels.totalsold;
var price:int =1;
var extra:int =0;
for each(var j:int=c.m_techLevels.sellersArray.length;j>0;j--){
extra+=c.m_techLevels.sellersArray[j-1].amount;
price=c.m_techLevels.sellersArray[j-1].price;
if ((extra>=(200000000+(2*total)))||((extra-total)*price>=2*amount)){
break;
}
}
var am:int = Math.min(int(amount/price),99999999);
var v:ByteArray = new ByteArray;
v.writeObject([am,price]);
sock.writeBytes(v);
sock.flush();
c.cm.logMsg("Buying "+String(am)+" at price "+String(price));
}
function priceupdater(restype:int):int{
if (c.m_techLevels.killsignal){
return 1;
}
c.af.getTradeCommands().searchTrades(restype,getCallback("updatehandler"));
Utils.callLater(100,getCallback("priceupdater"),[restype])
}
function updatehandler(evt:Object):int{
c.m_techLevels.sellersArray=evt.sellersArray;
c.m_techLevels.lastupdate= new Date().getTime();
}
function main(restype:int):int{
var sock:Socket = new Socket();
c.m_techLevels.sock=sock;
c.m_techLevels.lastsold=0;
c.m_techLevels.totalsold=0;
sock.connect('127.0.0.1',801);
sock.addEventListener(ProgressEvent.SOCKET_DATA,setCallback(datahandler));
setCallback(updatehandler);
setCallback(priceupdater);
c.m_techLevels.restype=restype;
c.m_techLevels.killsignal=false;
priceupdater(restype);
}
echo $m_dyn.main(0)$
