//-- Simple fast troop moving script v1.2,   by SeriousThisIsMe
//-- ATTENTION!! -> This script is for RE only (not for AE)
//-- This script will move all of the selected troops out of a town leaving only the
//-- entered amounts behind with no need to calculate troop amounts.
//-- 1) Copy this entire script to the "Script" tab under the town with the troops.
//-- 2) Enter troop types from the following:  wo w s p sw a c cata t b r cp
//--    **Be sure to enter the troops in the order you want them moved!**
//-- 3) Enter the amounts of those troops you would like to keep in this town.
//-- 4) Enter the coords of the destination to reinforce the troops to. You can enter
//--    coords for a single destination if all the troops are going to the same place,
//--    but if you want some troops to go to different places then you need to specify
//--    the destination coords for every troop type. The destinations can be any
//--    location that can normally be reinforced such as one of your towns, a valley
//--    owned by you (for quick valley transfer), an alliance member's town, or a
//--    valley owned by a member of your alliance.
//-- 5) Click "Run"


setarr sendTroopType "wo w s p sw a c cata t b r cp"
setarr howManyToKeep "2000 5000 2000 2000 2000 38500 2000 2000 22500 15000 100 100"
setarr sendToLocations "301,699"


//===============DO=NOT=CHANGE=BELOW=THIS=LINE===============
setsilence true
setarray troopIntNames ["peasants", "militia", "scouter", "pikemen", "swordsmen", "archer", "lightCavalry", "heavyCavalry", "carriage", "ballista", "batteringRam", "catapult"]
setarray troopExtNames ["Worker", "Warrior", "Scout", "Pikemen", "Swordsmen", "Archer", "Cavalry", "Cataphract", "Transporter", "Ballista", "BatteringRam", "Catapult"]
setarray troopAbbr ["wo", "w", "s", "p", "sw", "a", "c", "cata", "t", "b", "r", "cp"]
set t "allTroop"
set i 0
label checkDest
if {$c.getCityFromCoord(%sendToLocations[{%i%}]%)$==null} set t "troop"
set i {%i%+1}
if {%i%<%sendToLocations.length%} goto checkDest
set r getRallyDispatchPerArmyLimit
if {%sendTroopType.length%==%howManyToKeep.length%} goto isValidArrLength
echo "Error!  Number of values entered do not match in variables 'sendTroopType' and 'howManyToKeep'!"
echo "For every troop type entered you need to enter an amount to keep, even if it's 0."
goto exitScript
label isValidArrLength
if {%sendToLocations.length%==%sendTroopType.length%||%sendToLocations.length%==1} goto isValidDestinationArrLength
echo "Error!  Number of destination coordinates entered does not match the number of troop types entered!"
echo "You must either enter a single destination or a destination for each troop type."
goto exitScript
label isValidDestinationArrLength
echo "Begin moving troops..."
set totalTroopsMoved null
set troopStr empty
set currentTroopAmount 0
set sendTroopTypeIndex -1
label nextSendTroopType
set sendTroopTypeIndex {%sendTroopTypeIndex%+1}
if {!{%sendTroopType.length%>%sendTroopTypeIndex%}} goto doneSendingAllTroops
set indexTN -1
label findTroopName
set indexTN {%indexTN%+1}
if {%sendTroopType[{%sendTroopTypeIndex%}]%==%troopAbbr[{%indexTN%}]%} goto foundTroopName
if {{%indexTN%+1}==%troopAbbr.length%} goto couldNotFindTroopName
goto findTroopName
label couldNotFindTroopName
echo "WARNING!  Invalid value of ' %sendTroopType[{%sendTroopTypeIndex%}]% ' has been entered for variable 'sendTroopType'!"
echo "--'sendTroopType' is not among [ wo w s p sw a c cata t b r cp ]. Skipping to next troop type entered."
goto nextSendTroopType
label foundTroopName
if {{$c.cm.%t%.%troopIntNames[{%indexTN%}]%$-%howManyToKeep[{%sendTroopTypeIndex%}]%}>-1} goto howManyToKeepIsValid
echo "ATTENTION!  Invalid value of ' %howManyToKeep[{%sendTroopTypeIndex%}]% ' has been entered for troop type ' %sendTroopType[{%sendTroopTypeIndex%}]% '"
echo "--'howManyToKeep' is set to more than your current troop count. Skipping to next troop type entered."
goto nextSendTroopType
label howManyToKeepIsValid

label startSendingTroops
if {{$c.cm.%t%.%troopIntNames[{%indexTN%}]%$-%howManyToKeep[{%sendTroopTypeIndex%}]%+%currentTroopAmount%}>$c.cm.%r%(true)$} goto sendFull
if {{$c.cm.%t%.%troopIntNames[{%indexTN%}]%$-%howManyToKeep[{%sendTroopTypeIndex%}]%}<1&&%troopStr%==empty} goto doneSendingTroops
if {%troopStr%==empty} goto makeNewTroopStr
set troopStr "%troopStr%,%troopAbbr[{%indexTN%}]%:{$c.cm.%t%.%troopIntNames[{%indexTN%}]%$-%howManyToKeep[{%sendTroopTypeIndex%}]%}"

set isNextsendToLocations false
iferror %sendToLocations[{%sendTroopTypeIndex%+1}]% goto noNextsendToLocations

set isNextsendToLocations true
label noNextsendToLocations

label skipAddAnotherTroopType

label sendFull
if {%troopStr%==empty} goto makeNewFullTroopStr
set troopStr "%troopStr%,%troopAbbr[{%indexTN%}]%:{$c.cm.%r%(true)$-%currentTroopAmount%}"

label makeNewFullTroopStr
set troopStr "%troopAbbr[{%indexTN%}]%:$c.cm.%r%(true)$"

set currentTroopAmount 0
set troopStr empty
if {{$c.cm.%t%.%troopIntNames[{%indexTN%}]%$-%howManyToKeep[{%sendTroopTypeIndex%}]%}>0} goto startSendingTroops
label doneSendingTroops

if {%totalTroopsMoved%==null} goto noTroopsMoved
