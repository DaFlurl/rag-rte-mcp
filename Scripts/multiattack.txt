function rewardtimecheck(x) {
	var m=getStatic("herorewarddata");
	if (m==null) return true;
	if (m[x]==null) {
		m[x]=new Date().getTime();
		return true;
	}
	if (m[x]>((new Date().getTime())-305000)) return false;
	m[x]=new Date().getTime();
	return true;
}
function awardheroes() {
	for (var i=0;i<c.cm.heroes.length;i+=1) {
		if (c.cm.heroes[i].status==0 && c.cm.heroes[i].stamina >= 4 && rewardtimecheck(c.cm.heroes[i].id)) {
			c.cm.logMsg("Rewarding gold to "+c.cm.heroes[i].name);
			c.af.getHeroCommand().awardGold(c.castle.id,c.cm.heroes[i].id);
			c.af.getEquipmentCommands().WearAllEquipmenttech(c.cm.heroes[i].id,2);
		}
		else if (c.cm.heroes[i].status!=0||c.cm.heroes[i].isHero==0) continue;
		if (UseEnergyItems==1||c.cm.heroes[i].isHero!=0) {
			if (c.cm.heroes[i].stamina < 50 && c.getItem("hero.item.Tea").count >0) c.af.getHeroCommand().useItem(c.castle.id,c.cm.heroes[i].id,"hero.item.Tea");
			if (c.cm.heroes[i].stamina < 30 && c.getItem("hero.item.Lugui_Wine").count > 0) c.af.getHeroCommand().useItem(c.castle.id,c.cm.heroes[i].id,"hero.item.Lugui_Wine");
			if (c.cm.heroes[i].stamina < 10 && c.getItem("hero.item.Panax_ginseng").count > 0) c.af.getHeroCommand().useItem(c.castle.id,c.cm.heroes[i].id,"hero.item.Panax_ginseng");
		}
	}
}
function gettroops() {
	var s=[];
	var m,ok;
	var t=c.getValue("Troop",keeptroops).toObject();
	var i=0;
	while (i<preferredtroops.length) {
		m=c.getValue("Troop",preferredtroops[i]).toObject();
		s=c.cm.troop.toObject();
		ok=true;
		for (var p in m) {
			if (s[p]<m[p]) {
				ok=false;
				break;
			}
			s[p]-=m[p];
		}
		if (!ok) {
			i+=1;
			continue;
		}
		ok=true;
		for (var pp in m) {
			if (m[pp]==0) continue;
			if (s[pp]<t[pp]) {
				ok=false;
				break;
			}
		}
		if (!ok) {
			i+=1;
			continue;
		}
		return preferredtroops[i];
	}
	c.cm.logMsg("Couldn't find preferred troop");
	return "";
}
function armyevents(x) {
	if (x.hero.status!=0) {
		delete getStatic("armydata")[x.hero.id];
	}
	if (x.updateType==1) {
		var m=getStatic("herorewarddata");
		if (m!=null) {
			delete m[x.hero.id];
		}
		c.cm.logMsg("Deleting data for fired hero "+x.hero.name);
	}
}

function addarmyupdate() {
	if (getStatic("armydata")==null) {
		setStatic('armydata',{});
		if (getCallback("armyevents")==null) {
			ResponseDispatcher.getInstance().addEventListener(ResponseDispatcher.SERVER_HERO_UPDATE,setCallback(armyevents));
		}
	}
	if (getStatic("herorewarddata")==null) {
		setStatic("herorewarddata",{});
	}
}

function dorefresh() {
	if (c.lag>30) {
		if (getStatic("lastrefreshtime")!=null&&getStatic("lastrefreshtime")>((new Date().getTime())-360000)) return;
		setStatic("lastrefreshtime",new Date().getTime());
		MainScreen.getInstance().refresh();
	}
}

function fireheros() {
	for (var i=0;i<c.cm.heroes.length;i+=1) {
		if (c.cm.heroes[i].isHero == 0 && c.cm.heroes[i].stamina < 4 && c.cm.heroes[i].status == 0 &&  c.cm.heroes[i].level < 101) c.af.getHeroCommand().fireHero(c.castle.id,c.cm.heroes[i].id);
	}
}

function herolevelup() {
	var actions=0;
	for (var i=0;i<c.cm.heroes.length;i+=1) {
		if (c.cm.heroes[i].status!=0) continue;
		if ( c.cm.heroes[i].stamina < 4) continue;
		var l=c.cm.heroes[i].level;
		var exp=c.cm.heroes[i].experience;
		while (exp>c.cm.heroes[i].upgradeExp&&l<100) {
			c.af.getHeroCommand().levelUp(c.castle.id,c.cm.heroes[i].id);
			c.af.getHeroCommand().addPoint2(c.castle.id,c.cm.heroes[i].id,0,1,0);
			actions+=1;
			exp-=c.cm.heroes[i].upgradeExp;
			l+=1;
		} 
	}
	return int(actions*0.3);
}

function sendattacks(i) {
	if (c.cm.localArmies.length>=c.cm.getMaxRallySlots()) return i;
	if (getStatic("lastarmysendtime")!=null&&getStatic("lastarmysendtime")>((new Date().getTime())-Number(globalspeed)*1000)) {
		c.cm.logMsg("Rate limited, waiting");
		return i;
	}
	i=int(i);
	var jj=0;
	var ss=getStatic("armydata");
	var kk=new Date().getTime();
	for (;i<c.cm.heroes.length;i+=1) {
		if (c.cm.heroes[i].status!=0) continue;
		if (c.cm.heroes[i].stamina < 4) continue;
		if (c.cm.heroes[i].name=="Queen"&&queentofight!=1) continue;
		if (c.cm.heroes[i].name=="King"&&kingfight!=1) continue;
		var m;
		if (ss[c.cm.heroes[i].id]==null) {
			ss[c.cm.heroes[i].id]=kk+10000;
		}
		else if (ss[c.cm.heroes[i].id]>kk) {
			c.cm.logMsg("Retrying too fast with same hero");
			continue;
		}
		else {
			ss[c.cm.heroes[i].id]=ss[c.cm.heroes[i].id]+20000;
		}
		if (usebanners==1&&c.getItem("player.item.stygandrsbannerofthehorde").count>0) {
			m=c.getArmy("attack",c.getValue("Coordinate",coords),c.cm.heroes[i].name,c.getValue("Troop",bannertroops));
			m.useItem=true;
		}
		else {
			m=c.getArmy("attack",c.getValue("Coordinate",coords),c.cm.heroes[i].name,c.getValue("Troop",troops));
		}
		m.heroId=c.cm.heroes[i].id;
		c.af.getArmyCommands().newArmy(c.castle.id,m);
		c.cm.logMsg("Sending with "+c.cm.heroes[i].name);
		setStatic("lastarmysendtime",new Date().getTime());
		jj+=1;
		if (jj>=speed) break;
	}
	return i+1;
}

function fixstuckheroes() {
	var m=c.getRawTime();
	var s=false;
	for (var i=0;i<c.cm.localArmies.length;i+=1) {
		if (c.cm.localArmies[i].reachTime+2000<m) {
			c.af.getHeroCommand().callBackHero(c.castle.id,c.cm.localArmies[i].heroid);
			s=true;
		}
	}
	return s;
}

function hirebesthero() {
	var atk=0;
	var heroname=null;
	for (var i=0;i<c.cm.innHeroes.length;i+=1) {
		if (c.cm.innHeroes[i].power>atk) {
			atk=c.cm.innHeroes[i].power;
			heroname=c.cm.innHeroes[i].name;
		}
	}
	c.af.getHeroCommand().hireHero(c.castle.id,heroname);
	return heroname;
}

function isherospotsfull() {
	var max=c.cm.getBuildingLevel(c.buildingTypes.fh);
	for (var i=0;i<c.cm.heroes.length;i+=1) {
		if ((c.cm.heroes[i].name=="Queen"||c.cm.heroes[i].name=="King")&&c.cm.heroes[i].isHero!=0) {
			max+=1;
			break;
		}
	}
	if (c.cm.heroes.length >= max) return true;
	return false;
}
function getlag() {
	if (c.lag>10) {
		c.cm.logMsg("Lag "+c.lag);
		var mk=int(c.lag/2);
		if (mk>15) mk=15;
		return mk;
	}
	return 0;
}
