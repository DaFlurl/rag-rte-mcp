
set troopTypeToClear sw //use abrevations: wo w s p sw a c cata t b r cp
#
### DO NOT CHANGE BEYOND THIS LINE______________________________
setsilence true
setarray troopTypeIds ["null", "null", "wo", "w", "s", "p", "sw", "a", "t", "c", "cata", "b", "r", "cp"]
setarray troopNames ["null", "null", "workers", "warriors", "scouts", "pikemen", "swordsmen", "archers", "transporters", "cavalry", "cataphract", "ballistae", "batteringRams", "catapults"]
getindexof typeId troopTypeIds %troopTypeToClear%
set cqi 0
label cancelQueues
set qiv {$c.cm.troopProduceQueue[{%cqi%}].allProduceQueueArray.length$}
label cancelQueue
set qiv {%qiv%-1}
if {%qiv%<0} goto nextRack
if {$c.cm.troopProduceQueue[{%cqi%}].allProduceQueueArray[{%qiv%}].type$!=%typeId%} goto cancelQueue
label removeBatch
set checkValue $c.cm.troopProduceQueue[{%cqi%}].allProduceQueueArray[{%qiv%}].queueId$
set amt $c.cm.troopProduceQueue[{%cqi%}].allProduceQueueArray[{%qiv%}].num$
set r $c.af.getTroopCommands().cancelTroopProduce(c.castle.id,c.cm.troopProduceQueue[{%cqi%}].positionId,c.cm.troopProduceQueue[{%cqi%}].allProduceQueueArray[{%qiv%}].queueId)$
gosub waitRemoveResp %r% 10
iferror $%r%.ok$ goto removeBatch
if {$%r%.ok$!=1} echo $%r%.errorMsg$
if {$%r%.ok$!=1} goto removeBatch
label checkRemoved
set null $c.cm.updateTroopQueue()$
iferror $c.cm.troopProduceQueue[{%cqi%}].allProduceQueueArray[{%qiv%}].queueId$ goto batchCanceled
if {$c.cm.troopProduceQueue[{%cqi%}].allProduceQueueArray[{%qiv%}].queueId$!=%checkValue%} goto batchCanceled
sleep $c.lag$
goto checkRemoved
label batchCanceled
echo "Batch %qiv% containg %amt% %troopNames[{%typeId%}]% in barrack at positionId $c.cm.troopProduceQueue[{%cqi%}].positionId$ has been removed"
goto cancelQueue
label nextRack
set cqi {%cqi%+1}
if {%cqi%<$c.cm.troopProduceQueue.length$} goto cancelQueues
goto out
label waitRemoveResp
set _starttime $c.getRawTime()$
label waitRespStart
iferror {$%_1%.ok$} goto respWait
gosubreturn %_1%
label respWait
if {$c.getRawTime()$>{%_starttime%+%_2%*1000}} gosubreturn 0
sleep 0.1
goto waitRespStart
label out
set msg "<br><b>=== Trainig Queue Canceling by Troop Type ===</b><br><i>        Script created by &#169;&#60;&#126;Yozhik v Tumane&#126;&#62;&#8482;</i>"
set log $c.cm.logMsg(%msg%,#0000FF)$
