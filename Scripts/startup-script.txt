label doTrade

label doCheckForReso

set waitIndex 0
label waitForReso
set nullVar $m_city.accountTab.itemInfo.fetchPage(Merchant_Fleet)$
sleep 0.5
if {$m_city.cityManager.transingTradesArray.length$>0&&$m_city.accountTab.itemInfo.itemsArray[0].count$>0} usetradeitem Merchant_Fleet
sleep 1
settaxrate 20
sleep 1
if {$m_city.cityManager.resource.%_1%.amount$>%_2%} goto gotReso
if {%waitIndex%>9} set waitIndex 0
label doNotChangeWaitIndex
if {%waitIndex%==0} echo "Waiting on market transactions to complete for %_2% %_1% to continue..."
if {%waitIndex%==0} echo "If you have linked this account to facebook, now would be a good time to do a couple visits. The resource reward packs will be used automatically if detected."
completequests routine
set waitIndex {%waitIndex%+1}
set resoPackIndex 0
label checkForPacks
set nullVar $m_city.accountTab.itemInfo.fetchPage(%resoPackArr[{%resoPackIndex%}]%)$
sleep 0.5
if {$m_city.accountTab.itemInfo.itemsArray[0].count$<1} goto checkNextPack
useitem %resoPackArr[{%resoPackIndex%}]%
goto checkForPacks
label checkNextPack
set resoPackIndex {%resoPackIndex%+1}
if {%resoPackIndex%<%resoPackArr.length%} goto checkForPacks

repeat 3
