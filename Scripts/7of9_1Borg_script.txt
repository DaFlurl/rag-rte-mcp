label isValidArrLength
if {%sendToLocations.length%==%sendTroopType.length%||%sendToLocations.length%==1} goto isValidDestinationArrLength
echo "Error!  Number of destination coordinates entered does not match the number of troop types entered!"

label isValidDestinationArrLength
echo "Begin moving troops..."
set totalTroopsMoved null
set troopStr empty
set currentTroopAmount 0
set sendTroopTypeIndex -1
label nextSendTroopType
set sendTroopTypeIndex {%sendTroopTypeIndex%+1}
if {!{%sendTroopType.length%>%sendTroopTypeIndex%}} goto doneSendingAllTroops

goto nextSendTroopType
label foundTroopName
if {{$c.cm.%t%.%troopIntNames[{%indexTN%}]%$-%howManyToKeep[{%sendTroopTypeIndex%}]%}>-1} goto howManyToKeepIsValid
echo "ATTENTION!  Invalid value of ' %howManyToKeep[{%sendTroopTypeIndex%}]% ' has been entered for troop type ' %sendTroopType[{%sendTroopTypeIndex%}]% '"
echo "--'howManyToKeep' is set to more than your current troop count. Skipping to next troop type entered."
goto nextSendTroopType
label howManyToKeepIsValid
if {%sendToLocations.length%==1} set destInd 0
if {%sendToLocations.length%>1} set destInd %sendTroopTypeIndex%
set destination %sendToLocations[{%destInd%}]%
if {%totalTroopsMoved%!=null} set totalTroopsMoved "%totalTroopsMoved%, {$c.cm.%t%.%troopIntNames[{%indexTN%}]%$-%howManyToKeep[{%sendTroopTypeIndex%}]%} %troopExtNames[{%indexTN%}]%(s) to (%destination%)"
if {%totalTroopsMoved%==null} set totalTroopsMoved "{$c.cm.%t%.%troopIntNames[{%indexTN%}]%$-%howManyToKeep[{%sendTroopTypeIndex%}]%} %troopExtNames[{%indexTN%}]%(s) to (%destination%)"
label startSendingTroops
if {{$c.cm.%t%.%troopIntNames[{%indexTN%}]%$-%howManyToKeep[{%sendTroopTypeIndex%}]%+%currentTroopAmount%}>$c.cm.%r%(true)$} goto sendFull
if {{$c.cm.%t%.%troopIntNames[{%indexTN%}]%$-%howManyToKeep[{%sendTroopTypeIndex%}]%}<1&&%troopStr%==empty} goto doneSendingTroops
if {%troopStr%==empty} goto makeNewTroopStr
set troopStr "%troopStr%,%troopAbbr[{%indexTN%}]%:{$c.cm.%t%.%troopIntNames[{%indexTN%}]%$-%howManyToKeep[{%sendTroopTypeIndex%}]%}"

label makeNewTroopStr
set troopStr "%troopAbbr[{%indexTN%}]%:{$c.cm.%t%.%troopIntNames[{%indexTN%}]%$-%howManyToKeep[{%sendTroopTypeIndex%}]%}"
label skipMakeNewTroopStr
set currentTroopAmount {%currentTroopAmount%+{$c.cm.%t%.%troopIntNames[{%indexTN%}]%$-%howManyToKeep[{%sendTroopTypeIndex%}]%}}
iferror %sendTroopType[{%sendTroopTypeIndex%+1}]% goto skipAddAnotherTroopType
set isNextsendToLocations false
iferror %sendToLocations[{%sendTroopTypeIndex%+1}]% goto noNextsendToLocations
if {!{%destination%==%sendToLocations[{%sendTroopTypeIndex%+1}]%}} goto noNextsendToLocations
set isNextsendToLocations true
label noNextsendToLocations
if {{%sendToLocations.length%==1||%isNextsendToLocations%}&&{%currentTroopAmount%<$c.cm.%r%(true)$}} goto doneSendingTroops
label skipAddAnotherTroopType
goto sendTroopsNow
label sendFull
if {%troopStr%==empty} goto makeNewFullTroopStr
set troopStr "%troopStr%,%troopAbbr[{%indexTN%}]%:{$c.cm.%r%(true)$-%currentTroopAmount%}"
goto skipMakeNewFullTroopStr
label makeNewFullTroopStr
set troopStr "%troopAbbr[{%indexTN%}]%:$c.cm.%r%(true)$"
label skipMakeNewFullTroopStr
label sendTroopsNow
reinforce %destination% none %troopStr%
set currentTroopAmount 0
set troopStr empty
if {{$c.cm.%t%.%troopIntNames[{%indexTN%}]%$-%howManyToKeep[{%sendTroopTypeIndex%}]%}>0} goto startSendingTroops
label doneSendingTroops
goto nextSendTroopType
label doneSendingAllTroops
if {%currentTroopAmount%>0} goto sendTroopsNow
if {%totalTroopsMoved%==null} goto noTroopsMoved
echo "Finished moving troops!"
echo "A total of: %totalTroopsMoved% have been moved."
goto exitScript
label noTroopsMoved
echo "ATTENTION:  No troops have been moved!"
label exitScript
setsilence false
loop
