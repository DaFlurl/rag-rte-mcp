label honorFarmingsetup
include clearBotinfo

scanmap %x%,%y% %npcDistance%
set scanList "empty"
set yIndex {{%npcDistance%*-1}-1}
label checkNpc1row
set xIndex {%npcDistance%*-1}
set yIndex {%yIndex%+1}
if {%yIndex%>%npcDistance%} goto plusNpc1distance
label checkNpc1coords
set xIndex {%xIndex%+1}
if {%xIndex%>%npcDistance%} goto checkNpc1row
set xValue {%x%+{%xIndex%-1}}
if {%xValue%>=0} goto neg1Variablex
set xValue {%xValue%+$m_city.player.mapSizeX$}
label neg1Variablex
if {%xValue%<$m_city.player.mapSizeX$} goto size1Variablex
set xValue {%xValue%-$m_city.player.mapSizeX$}
label size1Variablex
set yValue {%y%+%yIndex%}
if {%yValue%>=0} goto neg1Variabley
set yValue {%yValue%+$m_city.player.mapSizeX$}
label neg1Variabley
if {%yValue%<$m_city.player.mapSizeX$} goto size1Variabley
set yValue {%yValue%-$m_city.player.mapSizeX$}
label size1Variabley
set mapCoord "{%xValue%},{%yValue%}"
set coordId $m_city.getValue(Coordinate,%mapCoord%)$
if {$m_city.getMapType(%coordId%)$!=12} goto checkNpc1coords
if {$m_city.getMapLevel(%coordId%)$!=1} goto checkNpc1coords
set npcCount {%npcCount%+1}
if {$%npcCount%$==1} goto setHonorone
if {$%npcCount%$==2} goto setHonortwo
label setHonorone
set firstnpc %mapCoord%
setarr npcList1 "%firstnpc% 2,1"
goto checkNpc1coords
label setHonortwo
set secondnpc %mapCoord%
setarr npcList1 "%firstnpc% %secondnpc%"
set attInterval 360
set farmIndex -1
goto sendHonortwo
label resendHonor
sleep 390
label sendHonortwo
set farmIndex -1
label sendHonorfarmer

if {{%farmIndex%+1}<%npcList1.length%} goto sendHonorfarmer
if {{$m_city.player.playerInfo.office$==General}&&{$m_city.player.playerInfo.titleId$>=4}} goto honorExitscript
if {{%honorTimestamp%+24}<={{$m_city.getRawTime()$-$m_city.player.playerInfo.createrTime$}/60000\60}} goto honorFarmingsetup
goto resendHonor
label plusNpc1distance
set npcDistance {%npcDistance%+1}
goto startNpc1farming

echo "collect 100 cav and 10k each resource to build next city"
if {$m_city.cityManager.troop.lightCavalry$<100} goto waitNewgridtroops
include openRespacks
echo "Ready for New City"
label gridMastersetup
set mainLocation $m_city.allCities[0].coord$
set mainId $m_city.getValue(Coordinate,%mainLocation%)$
set cityPlanid %mainId%
set x {{%cityPlanid%|$m_city.player.mapSizeX$}+%xVar%}
set y {{%cityPlanid%\$m_city.player.mapSizeX$}+%yVar%}
set gridCoord {%x%,%y%}
set typeGrid flat
set levelGrid 1
set radiusGrid 5
set centerGrid %gridCoord%
set stateStart $m_city.stateNameFromFieldId(m_city.getValue(Coordinate,m_city.allCities[0].coord))$
goto findGridtarget
label addGridlevel
set levelGrid {%levelGrid%+1}
if {%levelGrid%>2} goto noneFoundInState
goto findGridtarget
label noneFoundInState
echo "No new city flats in %stateStart% Limits"
set xVar {%xVar%/2}
set yVar {%yVar%/2}
goto gridMastersetup
label findGridtarget
scanmap $%centerGrid%$ %radiusGrid%
setarr gridCoordslist "$m_city.getMapFieldList(%typeGrid%,%levelGrid%,%radiusGrid%,m_city.getValue(Coordinate,%centerGrid%))$"
if {%gridCoordslist[0]%==null} goto addGridlevel
set gridIndex 0
label checkStategrid
if {%gridIndex%==%gridCoordslist.length%} goto noneFoundInState
set fieldId $m_city.getValue(Coordinate,%gridCoordslist[{%gridIndex%}]%)$
if {%stateStart%==$m_city.stateNameFromFieldId(m_city.getValue(Coordinate,%gridCoordslist[{%gridIndex%}]%))$} goto foundTarget
set gridIndex {%gridIndex%+1}
goto checkStategrid
label foundTarget
set newcityGrid "%gridCoordslist[{%gridIndex%}]%"
echo "closet flat lv%levelGrid% is at %newcityGrid% in %stateStart%"
config nomayor:1
setmayor remove
sleep 6
getspamhero
attack %newcityGrid% any cav:100

buildcity %newcityGrid%
echo "City number {$m_city.player.playerInfo.castleCount$+1} scheduled for creation at %newcityGrid%"
echo "Do not reset the bot until after city is completed"
goto exitNewgrid
label waitNewgridtroops
sleep 450
if {$m_city.cityManager.resource.gold$>100000} goto startNewgrid
include openRespacks
goto startNewgrid
label exitNewgrid

[startRosefarming]
label rosePrimarysetup
if {$m_city.castle.troop.archer$<100} goto waitRosetroop
include clearBotinfo
config hero:10,fasthero:20,herominattack:30,junkhero:10,nomayor:1
set roseTimestamp {{$m_city.getRawTime()$-$m_city.player.playerInfo.createrTime$}/60000}
set fh "Feasting Hall"
set rs "Rally Spot"
set i 0
set valleyDistance 1
label findValley
set mapArea $m_city.player.mapSizeX$
set fieldId $m_city.castle.fieldId$
set x {%fieldId%|%mapArea%}
set y {%fieldId%\%mapArea%}
scanmap %x%,%y% %valleyDistance%
set yIndex {{%valleyDistance%*-1}-1}
label checkRow
set xIndex {%valleyDistance%*-1}
set yIndex {%yIndex%+1}
ifgoto ( %yIndex% > %valleyDistance% ) scanComplete
label checkCoords
set xIndex {%xIndex%+1}
ifgoto ( {%xIndex%} > {%valleyDistance%} ) checkRow
set xValue {%x%+{%xIndex%-1}}
ifgoto ( {%xValue%} >= 0 ) xValueIsNotNeg
set xValue {%xValue%+%mapArea%}
label xValueIsNotNeg
ifgoto ( {%xValue%} < {%mapArea%} ) xValueIsNotTooBig
set xValue {%xValue%-%mapArea%}
label xValueIsNotTooBig
set yValue {%y%+%yIndex%}
ifgoto ( {%yValue%} >= 0 ) yValueIsNotNeg
set yValue {%yValue%+%mapArea%}
label yValueIsNotNeg
ifgoto ( {%yValue%} < {%mapArea%} ) yValueIsNotTooBig
set yValue {%yValue%-%mapArea%}
label yValueIsNotTooBig
set mapCoord "{%xValue%},{%yValue%}"
set coordId $m_city.getValue(Coordinate,%mapCoord%)$
if {$m_city.getMapType(%coordId%)$==10} goto checkValleylevel
ifgoto ( $m_city.getMapType(%coordId%)$ >= 6 ) checkCoords
label checkValleylevel
ifgoto ( $m_city.getMapLevel(%coordId%)$ >= 3 ) checkCoords
ifgoto ( {{true&&{{%xValue%}|18==9}&&{{%yValue%}|18==9}}||{true&&{{%xValue%}|66==33}&&{{%yValue%}|66==33}}||{true&&{{%xValue%}|200==100}&&{{%yValue%}|200==100}}||{true&&{{%xValue%}==0}&&{{%yValue%}==0}}} = true ) checkCoords
set roseTarget "%mapCoord%"
goto findBuildlevel
label scanComplete
set valleyDistance {%valleyDistance%+1}
goto findValley
label findBuildlevel
if {$m_city.castle.buildingsArray[{%i%}].name$==%fh%} set fhLevel {$m_city.castle.buildingsArray[{%i%}].level$}
if {$m_city.castle.buildingsArray[{%i%}].name$==%rs%} set rsLevel {$m_city.castle.buildingsArray[{%i%}].level$}
set i {%i%+1}
if {%i%=={$m_city.castle.buildingsArray.length$-1}} goto setAttackmax
goto findBuildlevel
label setAttackmax
if {%rsLevel%>=%fhLevel%} set waveMax %fhLevel%
if {%rsLevel%<=%fhLevel%} set waveMax %rsLevel%
label checkAttackout
abandon %roseTarget%
sleep 15
if {$m_city.allCities.indexOf(m_city)$+1==1} completequests routine
if {{$m_city.player.playerInfo.office$==Major}&&{$m_city.player.playerInfo.titleId$==2}} goto roseDone
if {$m_city.cityManager.getUsedRallySlots()$<%waveMax%} goto farmRoses
if {{{$m_city.getRawTime()$-$m_city.player.playerInfo.createrTime$}/60000\60}>{%roseTimestamp%+60}} goto rosePrimarysetup
goto checkAttackout
label farmRoses
getspamhero
sleep 3
if {$m_city.castle.troop.archer$<100} goto checkAttackout
attack %roseTarget% any a:100
goto checkAttackout
label waitRosetroop
sleep %cycle%
goto rosePrimarysetup
label roseDone

[countRespacks]
set incApack 0
set nullVar $m_city.accountTab.itemInfo.fetchPage(lesser_bonus_reward)$
set lBon $m_city.accountTab.itemInfo.itemsArray[0].count$
set nullVar $m_city.accountTab.itemInfo.fetchPage(medium_bonus_reward)$
set mBon $m_city.accountTab.itemInfo.itemsArray[0].count$
set nullVar $m_city.accountTab.itemInfo.fetchPage(greater_bonus_reward)$
set gBon $m_city.accountTab.itemInfo.itemsArray[0].count$
set nullVar $m_city.accountTab.itemInfo.fetchPage(lesser_resources_reward)$
set lRes $m_city.accountTab.itemInfo.itemsArray[0].count$
set nullVar $m_city.accountTab.itemInfo.fetchPage(medium_resources_reward)$
set mRes $m_city.accountTab.itemInfo.itemsArray[0].count$
set nullVar $m_city.accountTab.itemInfo.fetchPage(greater_resources_reward)$
set gRes $m_city.accountTab.itemInfo.itemsArray[0].count$
condgoal {{$m_city.castle.name$==Flat}&&{%incApack%==0}&&{%lBon%!=0}&&{%lBon%<30}&&{$m_city.cityManager.resource.gold$<100000}} include lesser1
condgoal {{$m_city.castle.name$==Flat}&&{%incApack%==0}&&{%lRes%!=0}&&{%lRes%<30}&&{$m_city.cityManager.resource.gold$<100000}} include lesser2
condgoal {{$m_city.castle.name$==Flat}&&{%incApack%==0}&&{%mBon%!=0}&&{%mBon%<30}&&{$m_city.cityManager.resource.gold$<100000}} include medium1
condgoal {{$m_city.castle.name$==Flat}&&{%incApack%==0}&&{%mRes%!=0}&&{%mRes%<30}&&{$m_city.cityManager.resource.gold$<100000}} include medium2
condgoal {{$m_city.castle.name$==Flat}&&{%incApack%==0}&&{%gBon%!=0}&&{%gBon%<30}&&{$m_city.cityManager.resource.gold$<100000}} include greater1
condgoal {{$m_city.castle.name$==Flat}&&{%incApack%==0}&&{%gRes%!=0}&&{%gRes%<30}&&{$m_city.cityManager.resource.gold$<100000}} include greater2
